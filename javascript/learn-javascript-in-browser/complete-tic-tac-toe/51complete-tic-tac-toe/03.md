```javascript

```

Since functions enable code reuse, you can refactor the previous code into functions like this:

```html
1<!-- Game Screen -->  
2<div id="gameScreen">
3  …
```

```css
1* {
2    margin: 0;
3    …
```

```javascript
1const gameBoard = document.querySelector(".game-board");
2const turnInfo = document.querySelector("#turnInfo");
3
4let currentPlayer = "X"; 
5
6// Add click event listener to parent (game board)
7gameBoard.addEventListener('click', (event) => {
8    const cell = event.target;
9
10    // Ensure the cell is empty before marking it
11    if (cell.textContent === "") {
12        gameLoop(cell);
13    }
14});
15
16const gameLoop = (cell) => {
17    
18    // Handle player move
19    handlePlayerMove(cell);
20
21    // Update turn info
22    updateTurnInfo();
23
24    // Switch players
25    currentPlayer = currentPlayer === "X" ? "O" : "X";
26};
27
28
29const handlePlayerMove = (cell) => {
30    cell.textContent = currentPlayer;
31};
32
33const updateTurnInfo = () => {
34    turnInfo.textContent = currentPlayer === "X" ? "Player 2" : "Player 1";
35};
```
