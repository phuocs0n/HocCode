## Modal Open and Close

Now, let's see all three closing paths working together.

```html
1<!DOCTYPE html>
2<html lang="en">
3<head>
4  <meta charset="UTF-8" />
5  <title>Modal Open and Close</title>
6  <link rel="stylesheet" href="style.css" />
7</head>
8<body>
9
10  <button id="open-modal">Open modal</button>
11
12  <div id="modal-overlay" class="overlay">
13    <div class="modal">
14      <p>Click outside or press Escape to close.</p>
15      <button class="modal-close">Close</button>
16    </div>
17  </div>
18
19  <script src="script.js"></script>
20</body>
21</html>
```

```css
1body {
2  font-family: sans-serif;

4}
5

7  position: fixed;
8  inset: 0;
```

```javascript
1var overlay = document.getElementById('modal-overlay');
2var modalBox = overlay ? overlay.querySelector('.modal') : null;
3var openBtn = document.getElementById('open-modal');
4var closeBtn = overlay ? overlay.querySelector('.modal-close') : null;
5
6var escapeHandler = null;
7
8function closeModal() {
9  if (!overlay) return;
10
11  overlay.classList.remove('modal-open');
12
13  if (escapeHandler) {
14    document.removeEventListener('keydown', escapeHandler);
15    escapeHandler = null;
16  }
17}
18
19function openModal() {
20  if (!overlay) return;
21
22  overlay.classList.add('modal-open');
23
24  escapeHandler = function (e) {
25    if (e.key === 'Escape') {
26      closeModal();
27    }
28  };
29
30  document.addEventListener('keydown', escapeHandler);
31}
32
33if (openBtn) {
34  openBtn.addEventListener('click', openModal);
35}
36
37if (closeBtn) {
38  closeBtn.addEventListener('click', closeModal);
39}
40
41if (overlay) {
42  overlay.addEventListener('click', function (e) {
43    if (e.target === overlay) {
44      closeModal();
45    }
46  });
47}
48
49if (modalBox) {
50  modalBox.addEventListener('click', function (e) {
51    e.stopPropagation();
52  });
53}
```

Click Run Code. Open the modal and try all three ways to close it.

- Click the close button
- Click outside the modal
- Press Escape

Every path calls the same closeModal() function. One function. Three triggers. Nothing duplicated.
