## Source Code: Linear Search

```python
1# function to perform  linear search
2def linear_search(lst, target):
3    
4    # traverse through each element
5    for index, element in enumerate(lst):
6        
7        # compare each element with a target value
8        if element == target:
9            return index
10    
11    # return None if the target isn't found in lst
12    return None
13
14lst = [9, 10, 5, 8, 7, 4, 11, 6, 15, 3]
15
16# set a target value
17target_value = 4
18
19result = linear_search(lst, target_value)
20
21if result:
22    print(f"Index of the target value: {result}")
23else:
24    print("Target value not found in the list.")
25
```
