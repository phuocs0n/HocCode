4        self.capacity = capacity

6       
7        self.front = 0  # Points to the item to be removed
8        self.rear = -1  # Points to the last inserted item
9        self.count = 0  # Number of elements currently in the queue
10
11
12    def is_full(self):
13        return self.count == self.capacity
14
15
16    def is_empty(self):
17        return self.count == 0
18
19
20    def enqueue(self, item):
21        if self.is_full():
22            print("We can't add elements. The queue is full.")
23            return
24       
25        self.rear = (self.rear + 1) % self.capacity
26        self.queue[self.rear] = item
27        self.count += 1
28
29
30    def dequeue(self):
31        if self.is_empty():
32            print("We can't remove elements. The queue is empty.")
33            return None
34       
35        item_removed = self.queue[self.front]
36        self.queue[self.front] = None
37        self.front = (self.front + 1) % self.capacity
38        self.count -= 1
39        return item_removed
40
41
42    def display(self):
43        print(self.queue)
44        print(f"Front index: {self.front}, Rear index: {self.rear}, Count: {self.count}")
