## Source Code: Create a Queue

```python
1# Create a class to represent a queue
2class Queue:
3    def __init__(self):
4        self.queue = []
5
6    # Add an item to the queue (enqueue)
7    def enqueue(self, item):
8        self.queue.append(item)
9
10    # Remove the front item from the queue (dequeue)
11    def dequeue(self):
12        return self.queue.pop(0)
13
14    # Print the current state of the queue with a message
15    def print_queue(self, message):
16        print(f"{message}: {self.queue}")
17
18# Initialize the queue
19queue1 = Queue()
20
21# Add elements to the queue
22queue1.enqueue(5)
23queue1.enqueue(10)
24queue1.enqueue(100)
25
26# Print the queue
27queue1.print_queue("Queue after adding 3 elements")
28
29# Remove one item
30print(f"Dequeued Element: {queue1.dequeue()}")
31queue1.print_queue("After removing one item")
32
33# Remove another item
34print(f"Dequeued Element: {queue1.dequeue()}")
35queue1.print_queue("After removing another item")
```

Output

```text
Queue after adding 3 elements: [5, 10, 100]
Dequeued Element: 5
After removing one item: [10, 100]
Dequeued Element: 10
After removing another item: [100]
```
