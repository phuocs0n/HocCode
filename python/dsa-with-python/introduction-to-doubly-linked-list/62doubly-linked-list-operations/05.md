44
45    new_node.prev = current

49def insert_node(self, data, position = None):
50    # check if the list is empty
51    # and append to empty list
52    if self.head is None:
53        self.append_to_empty(data)
54        return  # exit after handling this case
55
56    # if position is None or not provided
57    # append the node to the end
58    if position is None:
59        self.append_node(data)

61
62    # if position is less than 0 or greater than length, invalid position
63    if position <= 0 or position > self.length() + 1:

65        return
66    
67    # if position is 1, insert at the beginning
68    if position == 1:
69        self.insert_at_beginning(data)

71
72    # for a valid position,
73    # get the current length of the list
74    length = self.length()
75
76    # if the position exceeds the length
77    # append the node
78    if position > length:
79        self.append_node(data)

81
82    # if none of the above conditions are met,
83    # insert at the specified position

In this example, we forward traversed to our desired node. If necessary, you can perform backward traversal to achieve the same thing.

In large doubly linked lists, you may choose whether to perform forward or backward traversal depending on the position.
