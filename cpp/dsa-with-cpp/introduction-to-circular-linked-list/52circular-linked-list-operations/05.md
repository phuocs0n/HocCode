## Code: Append Element

Our function to append an element to a linked list will look like the following:

```cpp
// Add node into an empty list.
void append_into_empty(T data){
    // Create a new node with given data
    Node<T>* new_node = new Node<T>({data}); 
    // new node is the head
    head = new_node;
    // new node back to itself
    new_node->next = head;
}

// Add node to the end of the list
void append_node(T data){
    // If list is empty
    // call the helper to 
    // insert into empty list
    if(head == nullptr) {
        append_into_empty(data);
        return;
    }
    // Initially we are at the head node
    Node<T>* current = head;
    // Loop until the next node is a head
    while (current -> next != head){
        // Move to the next node.
        current = current->next;
    }
    // Create a new node with given data
    Node<T>* new_node = new Node<T>({data});
    // The next of the current node is new_node
    current->next = new_node;
    // new_node points to head
    new_node->next = head;
}
```
