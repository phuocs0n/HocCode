1#include <iostream>
2#include <vector>
3#include <algorithm>
4using namespace std;
5
6int maximize_profit(const vector<int>& w, const vector<int>& v, int capacity) {
7    // Initialize an empty DP table
8    vector<vector<int>> DP(w.size() + 1, vector<int>(capacity + 1, 0));
9
10    // Run the loop once for each item
11    for (size_t item_index = 0; item_index < w.size(); ++item_index) {
12        // Compare and change values
13        for (int index = 0; index <= capacity; ++index) {
14            // You can only add an item if the bag size
15            // is larger than the item weight
16            if (index >= w[item_index]) {
17                DP[item_index + 1][index] = max(
18                    DP[item_index][index], 
19                    DP[item_index][index - w[item_index]] + v[item_index]
20                );
21            } 
22            // Otherwise, just copy the profit from one row above
23            else {
24                DP[item_index + 1][index] = DP[item_index][index];
25            }
26        }
27    }
28
29    // The last cell returns the maximum profit.
30    return DP[w.size()][capacity];
31}
32
33int main() {
34    // Initialize variables
35    vector<int> w = {2, 3, 6};
36    vector<int> v = {60, 100, 150};
37    int capacity = 5;
38
39    // Call the function and print the result
40    cout << maximize_profit(w, v, capacity) << endl;
41
42    return 0;
43}
