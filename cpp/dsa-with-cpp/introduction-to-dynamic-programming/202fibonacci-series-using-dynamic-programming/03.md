## Fibonacci Series Using Memoization

Here's how we can generate the Fibonacci series using memoization.

```cpp
1#include <iostream>
2#include <unordered_map>
3using namespace std;
4
5unordered_map<int, int> fib = {{0, 0}, {1, 1}};
6
7int Fibonacci(int n) {
8    // If we already have the result, just use it
9    if (fib.find(n) != fib.end()) {
10        return fib[n];
11    }
12    // Otherwise compute
13    fib[n] = Fibonacci(n - 1) + Fibonacci(n - 2);
14    return fib[n];
15}
16
17int main() {
18    int result = Fibonacci(4);
19    cout << result << endl;
20
21    return 0;
22}
```
