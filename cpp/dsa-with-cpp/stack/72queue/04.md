1#include <iostream>
2
3using namespace std;
4
5// Template structure for a Node in a linked list
6template <typename T>
7struct Node {
8    T data;                   // Data value of the node
9    Node<T>* next = nullptr;  // Pointer to the next node in the list
10};
11
12// Template class for handling a linked list
13template <typename T>
14class LinkedList {
15  public:
16    // Constructor initializes the linked list as empty
17    LinkedList() = default;
18
19    // Destructor deallocates all nodes to prevent memory leaks
20    ~LinkedList() {
21        Node<T>* current = head;
22        while (current) {
23            Node<T>* next = current->next;  // Save next node
24            delete current;                 // Delete the current node
25            current = next;                 // Move to next node
26        }
27        // Ensure the head is nullptr after destruction
28        head = nullptr;  
29        count = 0;
30    }
31
32    // Deletes the first node of the list
33    void delete_first_node() {
34        Node<T>* node_to_delete = head;  
35        // Head is moved to the second node
36        head = head->next;
37        // Delete the former head node
38        delete node_to_delete;  
39        // Decrement the count of nodes
40        --count;
41    }
42
43     // Returns the first element of the linked list
44     T get_first_element() {
45        return head->data;
46     }
47
48    // Add a node to the end of the list
49    void insert_node_at_end(T data) {
50        Node<T>* new_node = new Node<T>({data});  // Create a new node
51        
52        // If list is empty, 
53        // new node is the head
54        if (head == nullptr) {  
55            head = new_node;
56            tail = new_node;
57            ++count;
58            return;
59        }
60        
61        // Link the last node to the new node
62        tail->next = new_node;
63        
64        // Update the tail node with the new node
65        tail = tail->next;
66
67        // Increment the count of nodes
68        ++count;
69    }
70
71
72      // Returns the count of nodes in the linked list
73      int count_elements() {
74        return count;
75     }
76
77    // Displays the data in the list
78    void display() {
79        Node<T>* current = head;
80        // Traverse the list and print each node's data
81        while (current != nullptr) {
82            cout << current->data << " ";
83            current = current->next;
84        }
85        cout << "\n";  // End the line
86    }
87
88  private:
89    Node<T>* head {nullptr}; // Pointer to the head node
90    Node<T>* tail {nullptr}; // Pointer to the tail node
91    int count {0}; // Count of the number of nodes
92};
93
94template< typename T>
95class Queue {
96  public:
97    Queue() = default; // Defaulted constructor
98    ~Queue() = default; // Defaulted destructor
99
100    // Check if the queue is empty
101    bool is_empty() {
102        return (linked_list.count_elements() == 0);
103    }
104
105    // Enqueue an element to the queue
106    void enqueue(T item) {
107        linked_list.insert_node_at_end(item);
108    }
109
110    // Dequeue an element from the queue     
111    void dequeue() {
112        if(!is_empty())
113            linked_list.delete_first_node();
114    }
115  
116    // Get the first element of the queue
117    T peek() {
118        return linked_list.get_first_element();
119    }
120
121    // Display all elements in the queue
122    void print_queue() {
123        linked_list.display();
124    }
125  private:
126    LinkedList<T> linked_list; // Underlying linked list to store queue elements
127};
128
129int main() {
130    Queue <int> queue1;
131    
132    // Enqueue elements to queue
133    queue1.enqueue(5);
134    queue1.enqueue(10);
135    queue1.enqueue(100);
136    
137    // Display all the elements in the queue
138    cout << "Queue after adding three elements: ";
139    queue1.print_queue();
140    
141    // Dequeue one element from the queue
142    queue1.dequeue();
143
144    // Display all the elements in the queue
145    cout << "Queue after removing one element: ";
146    queue1.print_queue();
147
148    // Peek the first element of the queue
149    cout << "Peek the queue: " << queue1.peek() << '\n';
150
151// Display all the elements in the queue
152    cout << "Queue after peek operation: ";
153    queue1.print_queue();
154
155    return 0;
156}
