1// Member function to check if the linked list is sorted in non-decreasing order
2bool is_sorted(){
3    // An empty linked list is considered sorted
4    if (head == nullptr) return true;
5    
6    // Start from the head of the list
7    Node<T>* current = head;
8    
9    // Continue to traverse as long as there is a next node
10    while(current->next != nullptr){
11        // If the current node's data is greater than the next node's,
12        // the list is not sorted
13        if (current->data > current->next->data) return false;
14        
15        // Move to the next node in the list
16        current = current->next;
17    }
18    // If no unsorted pairs found, the list is sorted
19    return true;
20}
