## Add Two Distances using Struct

In this example, we will learn to take two distances (in the inch-feet system), add them and display the result on the screen.

### Source Code

```cpp
1#include <iostream>
2using namespace std;
3
4typedef struct Distance {
5    int feet;
6    int inch;
7} distances;
8
9int main() {
10
11    // create struct variables
12    distances d1;
13    distances d2;
14    distances result;
15
16    // get feet and inch inputs for d1 and d2
17
18    cout << "Enter 1st distance" << endl;
19    cout << "Enter feet and inch: ";
20
21    cin >> d1.feet >> d1.inch;
22
23    cout << "Enter 2nd distance" << endl;
24    cout << "Enter feet and inch: ";
25
26    cin >> d2.feet >> d2.inch;
27
28    // find sum
29    result.feet = d1.feet + d2.feet;
30    result.inch = d1.inch + d2.inch;
31
32    // if inch is greater than 12 add it to feet
33    // and reduce inch
34    result.feet = result.feet + (result.inch / 12);
35    result.inch = result.inch % 12;
36
37    cout << "Sum: " << result.feet << " feet and " << result.inch << " inch";
38
39    return 0;
40}
```
