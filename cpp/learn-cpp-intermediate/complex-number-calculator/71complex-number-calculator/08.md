## Step 6: Completing the Program

Now, let's complete this program by combining all the components we've developed so far.

```cpp
1#include <iostream>
2#include <cmath>
3using namespace std;
4
5class Complex {
6private:
7
8    // attributes
9    double real;
10    double imag;
11
12public:
13    
14    // constructor to initialize variables with default values
15    Complex() : real(0.0), imag(0.0) {}
16    
17    // constructor to initialize variables with custom values
18    Complex(double r, double i) : real(r), imag(i) {}
19    
20    // overload the + operator
21    Complex operator + (Complex arg) {
22        
23        Complex complex_sum;
24        
25        // add the real and imaginary parts separately
26        complex_sum.real = this->real + arg.real;
27        complex_sum.imag = this->imag + arg.imag;
28        
29        return complex_sum;
30    }
31    
32    // overload the - operator
33    Complex operator - (Complex arg) {
34        
35        Complex complex_difference;
36        
37        // subtract the real and imaginary parts
38        complex_difference.real = this->real - arg.real;
39        complex_difference.imag = this->imag - arg.imag;
40        
41        return complex_difference;
42    }
43    
44    // overload the * operator
45    Complex operator * (Complex arg) {
46        
47        Complex complex_product;
48        
49        // use complex multiplication formula
50        complex_product.real = (this->real * arg.real) - (this->imag * arg.imag);
51        complex_product.imag = (this->real * arg.imag) + (this->imag * arg.real);
52        
53        return complex_product;
54    }
55    
56
57    // overload the / operator
58    Complex operator / (Complex arg) {
59
60        Complex complex_division;
61
62        try {
63
64            double denominator = pow(arg.real, 2) + pow(arg.imag, 2);
65        
66            // throw an exception if denominator is 0    
67            if (denominator == 0)
68                throw 0;
69
70            // use complex division formula
71            complex_division.real = ((this->real * arg.real) + (this->imag * arg.imag)) / denominator;
72            complex_division.imag = ((this->imag * arg.real) - (this->real * arg.imag)) / denominator;
73        }
74
75        catch (int num) {
76
77            // print the error message and exit the program
78            cout << "Error! Cannot divide by " << num << endl;
79            exit(0);
80        }
81        
82        return complex_division;
83    }
84
85    
86    // method to return the result
87    string get_result() {
88    
89        string result;
90       
91        if (imag == 0) {
92            result = to_string(real);
93        }
94        else if (imag < 0) {
95            result = to_string(real) + " - " + to_string(abs(imag)) + "i";
96        }
97        else {
98            result = to_string(real) + " + " + to_string(imag) + "i";
99        }
100        
101        return result;
102    }
103};
104
105int main() {
106    
107    Complex complex1(5, -10);
108    Complex complex2(0, 0);
109    
110
111    // divide complex1 by complex2
112    Complex division = complex1 / complex2;
113
114    cout << "Division = " << division.get_result() << endl;
115
116    
117    // subtract complex2 from complex1
118    Complex difference = complex1 - complex2;
119
120    cout << "Difference = " << difference.get_result() << endl;
121
122    return 0;
123}
```

Congratulations! You've completed the basic program for complex calculation!

As you can see, we first performed a complex number division by 0, followed by complex subtraction.

But thanks to our exception handling, the program prints and error message and exits the program the moment it encounters the illegal division.
