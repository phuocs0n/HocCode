## Step 4: Adding the main() Function

Now, let's add the main() function and test our Complex class. In main(), we will:

- Create two objects complex1 and complex2.
- Add the two objects and display the result.
- Then, multiply the two objects and display the result.

#### Source Code

```cpp
1#include <iostream>
2
3#include <cmath>
4
5using namespace std;
6
7class Complex {
8private:
9
10    // attributes
11    double real;
12    double imag;
13
14public:
15    
16    // constructor to initialize variables with default values
17    Complex() : real(0.0), imag(0.0) {}
18    
19    // constructor to initialize variables with custom values
20    Complex(double r, double i) : real(r), imag(i) {}
21    
22    // overload the + operator
23    Complex operator + (Complex arg) {
24        
25        Complex complex_sum;
26        
27        // add the real and imaginary parts separately
28        complex_sum.real = this->real + arg.real;
29        complex_sum.imag = this->imag + arg.imag;
30        
31        return complex_sum;
32    }
33    
34    // overload the - operator
35    Complex operator - (Complex arg) {
36        
37        Complex complex_difference;
38        
39        // subtract the real and imaginary parts
40        complex_difference.real = this->real - arg.real;
41        complex_difference.imag = this->imag - arg.imag;
42        
43        return complex_difference;
44    }
45    
46    // overload the * operator
47    Complex operator * (Complex arg) {
48        
49        Complex complex_product;
50        
51        // use complex multiplication formula
52        complex_product.real = (this->real * arg.real) - (this->imag * arg.imag);
53        complex_product.imag = (this->real * arg.imag) + (this->imag * arg.real);
54        
55        return complex_product;
56    }
57    
58    // overload the / operator
59    Complex operator / (Complex arg) {
60        
61        Complex complex_division;
62        double denominator = pow(arg.real, 2) + pow(arg.imag, 2);
63            
64        // use complex division formula
65        complex_division.real = ((this->real * arg.real) + (this->imag * arg.imag)) / denominator;
66        complex_division.imag = ((this->imag * arg.real) - (this->real * arg.imag)) / denominator;
67        
68        return complex_division;
69    }
70    
71    // method to return the result
72    string get_result() {
73    
74        string result;
75       
76        if (imag == 0) {
77            result = to_string(real);
78        }
79        else if (imag < 0) {
80            result = to_string(real) + " - " + to_string(abs(imag)) + "i";
81        }
82        else {
83            result = to_string(real) + " + " + to_string(imag) + "i";
84        }
85        
86        return result;
87    }
88};
89
90
91int main() {
92    
93    Complex complex1(5, -10);
94    Complex complex2(3, 6);
95    
96    // add complex1 and complex2
97    Complex total = complex1 + complex2;
98
99    cout << "Sum = " << total.get_result() << endl;
100    
101    // multiply complex1 and complex2
102    Complex product = complex1 * complex2;
103
104    cout << "Product = " << product.get_result() << endl;
105    
106    return 0;
107}
```

Output

```text
Sum = 8.000000 - 4.000000i
Product = 75.000000
```

Here, we have used the overloaded + and * operators to add and multiply the complex1 and complex2 objects.

Notice that our output consists of decimal numbers containing 6 digits after the decimal point. This is the default format in which the to_string() function converts double values.

While we can fix the format of this output, it is beyond the scope of this course. Instead, we will focus on handling an obvious exception that can occur during division.
