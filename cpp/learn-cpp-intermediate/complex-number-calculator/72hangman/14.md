## Step 1.9: Putting Everything Together

Let's combine all the components and create a working game of hangman.

```cpp
1#include <iostream>
2using namespace std;
3
4// max_misses always has a value of 6
5const int max_misses = 6;
6
7class Message {
8public:
9
10    // method to greet the player
11    void greet() {
12        cout << "------------------------------" << endl;
13        cout << "\t\tH A N G M A N" << endl;
14        cout << "------------------------------" << endl;
15        cout << "Guess the Word" << endl << endl;
16    }
17
18    // method to display the number of misses
19    void display_misses(int misses) {
20        cout << "Remaining misses: "<< max_misses - misses << endl << endl;
21    }
22
23    // method to display the incorrect guesses
24    void display_incorrect_guesses(string incorrect) {
25        cout << "Incorrect Guesses: " << incorrect << endl;
26    }
27    
28    // method to display the user answer
29    void display_user_answer(string user_answer) {
30        cout << "Your Answer: ";
31        for (char c: user_answer) {
32            cout << c << " ";
33        }
34
35        cout << endl;
36    }
37
38    // method to call when the user
39    // enters non-alphabetic character
40    void invalid_character_message() {
41        cout << "\nInvalid character!" << endl;
42        cout << "Input must be an alphabet!" << endl;
43    }
44
45    // method to call when the user enters a
46    // character that has already been entered
47    void already_entered_message() {
48        cout << "\nYou have already entered this letter!" << endl;
49        cout << "Enter another letter!" << endl;
50    }
51
52    void correct_message() {
53        cout << "\nCorrect!" << endl;
54    }
55
56    void incorrect_message() {
57        cout << "\nIncorrect!" << endl;
58    }
59
60    // print win or loss message
61    void final_result(string user_answer, string word) {
62        cout << "Your Answer: ";
63        for (char c: user_answer) {
64            cout << c << " ";
65        }
66
67        cout << endl << endl;
68
69        // print win message
70        if (user_answer == word) {
71            cout << "-----------------------------------------------" << endl;
72            cout << "\t\t\t\tY O U  W I N" << endl;
73            cout << "-----------------------------------------------" << endl;
74        }
75
76        // print loss message
77        else {
78            cout << "The correct answer is: " << word << endl << endl;
79            cout << "-----------------------------------------------" << endl;
80            cout << "\t\t\t\tY O U  L O S E" << endl;
81            cout << "-----------------------------------------------" << endl;
82        }
83    }
84};
85
86class Game {
87private:
88
89    string word;
90    string user_answer;
91    string incorrect;
92    string entered_characters;
93    int misses;
94    Message message;
95
96public:
97
98    // constructor to initialize attributes
99    Game(string chosen_word) {
100        word = chosen_word;
101        incorrect = "";
102        entered_characters = "";
103        misses = 0;
104
105        for (int i = 0; i < word.length(); ++i) {
106            user_answer += "_";
107        }
108    }
109
110    // method to ensure that user input is
111    // an alphabet and not a previous input
112    bool is_guess_valid(char letter) {
113
114        if (!isalpha(letter)) {
115            message.invalid_character_message();
116            return false;
117        }
118
119        for (int i = 0; i < entered_characters.length(); ++i) {            
120            if (letter == entered_characters[i]) {
121                message.already_entered_message();
122                return false;
123            }
124        }
125
126        return true;
127    }
128
129    // method to check if guess is correct
130    void is_guess_correct(char letter) {
131
132        bool guess = false;
133
134        for (int i = 0; i < word.length(); i++) {
135            if (letter == word[i]) {
136                user_answer[i] = letter;
137                guess = true;
138            }
139        }
140
141        if (guess) {
142            message.correct_message();
143        }
144        else {
145            message.incorrect_message();
146            incorrect += letter;
147            incorrect += ", ";
148            ++misses;
149        }
150
151        entered_characters += letter;
152    }
153
154    // method to check if the game is over
155    bool is_game_over() {
156        if (user_answer == word || misses >= max_misses) {
157            return true;
158        }
159        return false;
160    }
161
162    // method to display the game status
163    void display_game_status() {
164        message.display_incorrect_guesses(incorrect);
165        message.display_user_answer(user_answer);
166        message.display_misses(misses);
167    }
168
169    // method to display the final result
170    void display_final_result() {
171        message.final_result(user_answer, word);
172    }
173};
174
175int main() {
176
177    // chose the mystery word
178    string chosen_word = "ALGORITHM";
179
180    // variable to take user input
181    char letter;
182
183    // create message object
184    Message message;
185
186    // create Game object and
187    // pass chosen_word to its constructor
188    Game game(chosen_word);
189
190    // greet the player
191    message.greet();
192
193    // play game till the game is over
194    while (!game.is_game_over()) {
195
196        // display the game status
197        game.display_game_status();
198        
199        cout << "\nPlease enter your guess: ";
200
201        // get user input
202        cin >> letter;
203    
204        // convert letter to uppercase
205        letter = toupper(letter);
206
207        // check if input is an alphabet
208        if (game.is_guess_valid(letter)) {
209
210            // if input is valid, check if guess is correct
211            game.is_guess_correct(letter);
212        }
213    }
214
215    game.display_final_result();
216
217    return 0;
218}
```

- loss condition

The output should display
