10// Message class for displaying messages and design

90// Game class to code the game logic
91class Game {
92private:
93
94    string word;
95    string user_answer;
96    string incorrect;
97    string entered_characters;
98    int misses;
99    Message message;
100
101public:
102
103    // constructor to initialize attributes
104    Game(string chosen_word) {
105        word = chosen_word;
106        incorrect = "";
107        entered_characters = "";
108        misses = 0;
109
110        for (int i = 0; i < word.length(); ++i) {
111            user_answer += "_";
112        }
113    }
114
115    // method to ensure that user input is
116    // an alphabet and not a previous input
117    bool is_guess_valid(char letter) {
118
119        if (!isalpha(letter)) {
120            message.invalid_character_message();
121            return false;
122        }
123
124        for (int i = 0; i < entered_characters.length(); ++i) {            
125            if (letter == entered_characters[i]) {
126                message.already_entered_message();
127                return false;
128            }
129        }
130
131        return true;
132    }
133
134    // method to check if guess is correct
135    void is_guess_correct(char letter) {
136
137        bool guess = false;
138
139        for (int i = 0; i < word.length(); i++) {
140            if (letter == word[i]) {
141                user_answer[i] = letter;
142                guess = true;
143            }
144        }
145
146        if (guess) {
147            message.correct_message();
148        }
149        else {
150            message.incorrect_message();
151            incorrect += letter;
152            incorrect += ", ";
153            ++misses;
154        }
155
156        entered_characters += letter;
157    }
158
159    // method to check if the game is over
160    bool is_game_over() {
161        if (user_answer == word || misses >= max_misses) {
162            return true;
163        }
164        return false;
165    }
166
167    // method to display the game status
168    void display_game_status() {
169        message.display_incorrect_guesses(incorrect);
170        message.display_user_answer(user_answer);
171        message.display_misses(misses);
172    }
173
174    // method to display the final result
175    void display_final_result() {
176        message.final_result(user_answer, word);
177    }
178};
179
