## Example: C++ Pointer to Object

```cpp
1#include <iostream>
2using namespace std;
3
4class Student {
5    public:
6        double marks;
7};
8
9int main() {
10
11    Student student1;
12
13
14    // create Student pointer
15    // assign address of student1 object to it
16    Student* ptr = &student1;
17
18
19    // set marks of student1 to 66.6
20    student1.marks = 66.6;
21
22
23    // print marks using pointer
```
