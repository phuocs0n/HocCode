1#include <iostream>

6class Game {

12    public:

15        string get_computer_pick() {
16

18            srand(time(NULL));

21            int random_number = (rand() % 3) + 1;

23            switch(random_number) {

27        
28                case 2:

41        string get_user_pick() {
42
43            char user_input;
44
45            // get input until user enters 'r', 'p', or 's'
46            while (1) {
47
48                cout << "Enter Your Choice (r for rock, p for paper, s for scissors): ";
49                cin >> user_input;
50
51                // convert user_pick to lowercase
52                user_input = tolower(user_input);
53
54                // assign appropriate string to user_pick
55                if (user_input == 'r') {
56                    user_pick = "rock";
57
58                    // terminate while loop

63

68                    user_pick = "scissors";

80
81            // condition for user to draw

84            }

87            else if (user_pick == "paper" && computer_pick == "rock") {

96                result = "win";
