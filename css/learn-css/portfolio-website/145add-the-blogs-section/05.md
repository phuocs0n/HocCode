1/* Blog Section Styling - Same as Hero */
2.blogs {

12
13/* Card Container */
14.card-row {
15    display: flex;
16    flex-wrap: wrap;    /* Allow the cards to wrap on smaller screens */

23
24/* Blog card styling */
