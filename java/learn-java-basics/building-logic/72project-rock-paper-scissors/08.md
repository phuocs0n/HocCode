## Putting It All Together

```java
1import java.util.Scanner;
2
3class Main {
4
5    // Returns either "rock", "paper" or "scissors"
6    String getComputerChoice() {
7
8        String computerChoice;
9        int randomNumber = (int) (Math.random() * 3) + 1;
10
11        if (randomNumber == 1) {
12            computerChoice = "rock";
13        }
14
15        else if (randomNumber == 2) {
16            computerChoice = "paper";
17        }
18
19        else {
20            computerChoice = "scissors";
21        }
22
23        return computerChoice;
24    }
25
26    // Returns user input
27    String getUserInput() {
28
29        Scanner input = new Scanner(System.in);
30
31        System.out.println("Enter rock, paper or scissors:");
32        String userInput = input.nextLine();
33
34        userInput = userInput.toLowerCase();
35
36        return userInput;
37    }
38
39    // Returns either "win", "lose" or "draw"
40    String getResult(String userInput, String computerChoice) {
41
42        // Condition for user to draw
43        if (userInput.equals(computerChoice)) {
44            return "draw";
45        }
46
47        // Condition for user to win
48        else if (userInput.equals("paper") && computerChoice.equals("rock")) {
49            return "win";
50        } 
51        
52        else if (userInput.equals("rock") && computerChoice.equals("scissors")) {
53            return "win";
54        }
55        
56        else if (userInput.equals("scissors") && computerChoice.equals("paper")) {
57            return "win";
58        }
59        
60        // All other conditions result to loss
61        else {
62            return "lose";
63        }
64    }
65
66    public static void main(String[] args) {
67
68        // Creating object of Main class
69        Main obj = new Main();
70
71        // Get users input
72        String userInput = obj.getUserInput();
73        
74        // Get computers choice
75        String computerChoice = obj.getComputerChoice();
76
77        // Get results
78        String result = obj.getResult(userInput, computerChoice);
79
80        System.out.println("User Input: " + userInput);
81        System.out.println("Computer Choice: " + computerChoice);
82        System.out.println(result);
83    }
84}
```
