### Project Changes

- created another class Main with the main() method
- created an object of the Game class
- called getComputerPick(), getUserPick(), getResult() methods.

```java
1import java.util.Scanner;
2
3class Game {
4
5    String getComputerPick() {
6
7        String computerPick;
8        // get an integer from 1 to 3
9        int randomNumber = (int) (Math.random() * 3) + 1;
10
11        if (randomNumber == 1) {
12            computerPick = "rock";
13        }
14
15        else if (randomNumber == 2) {
16            computerPick = "paper";
17        }
18
19        else {
20            computerPick = "scissors";
21        }
22
23        return computerPick;
24    }
25
26    String getUserPick() {
27
28        Scanner input = new Scanner(System.in);
29
30        String userPick;
31
32        // get input until user enters "rock", "paper" or "scissors"
33        while (true) {
34            System.out.println("Enter rock, paper or scissors:");
35            userPick = input.nextLine();
36            if (userPick.equals("rock") || userPick.equals("paper") || userPick.equals("scissors")) {
37                break;
38            }
39        }
40
41        return userPick;
42    }
43
44    // return either "win", "lose" or "draw"
45    String getResult(String userPick, String computerPick) {
46
47        // condition for user to draw
48        if (userPick.equals(computerPick)) {
49            return "draw";
50        }
51
52        // condition for user to win
53        else if (userPick.equals("paper") && computerPick.equals("rock")) {
54            return "win";
55        }
56
57        else if (userPick.equals("rock") && computerPick.equals("scissors")) {
58            return "win";
59        }
60
61        else if (userPick.equals("scissors") && computerPick.equals("paper")) {
62            return "win";
63        }
64
65        // all conditions results to loss
66        else {
67            return "lose";
68        }
69    }
70}
71
72class Main {
73    public static void main(String[] args) {
74
75        Game game = new Game();
76        String userPick = game.getUserPick();
77        String computerPick = game.getComputerPick();
78        String result = game.getResult(userPick, computerPick);
79
80        System.out.println("User Pick: " + userPick);
81        System.out.println("Computer Pick: " + computerPick);
82        System.out.println("You " + result);
83
84    }
85}
```
