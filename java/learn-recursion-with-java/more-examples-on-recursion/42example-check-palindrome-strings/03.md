```java
1class Main {
2
3    public static boolean checkPalindrome(String text) {
4
5        // If the string is empty or contains a single character,
6        // the string is a palindrome
7        if (text.isEmpty() || text.length() == 1) {
8            return true;
9        }
10        
11        // Check if the first and last characters are the same
12        if (text.charAt(0) == text.charAt(text.length() - 1)) {
13
14            // Recursive call
15            return checkPalindrome(text.substring(1, text.length() - 1));
16        }
17        else {
18            return false;
19        }
20    }
21
22    public static void main(String[] args) {
23
24        boolean isPalindrome = checkPalindrome("level");
25
26        // Use ternary operator to determine result
```
