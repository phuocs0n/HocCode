```undefined
1import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
2import { useState } from 'react'
3import { fetchTasks, fetchTask, createTask, updateTask, deleteTask } from './api'
4import { taskKeys } from './queryKeys'
5 
6export default function TaskList() {
7  const [title, setTitle] = useState('')
8  const [status, setStatus] = useState('')
9  const queryClient = useQueryClient()
10 
11  const { data: tasks, isLoading, isError } = useQuery({
12    queryKey: taskKeys.list({ status }),
13    queryFn: () => fetchTasks({ status }),
14  })
15 
16  function handleMouseEnter(taskId) {
17    queryClient.prefetchQuery({
18      queryKey: taskKeys.detail(taskId),
19      queryFn: () => fetchTask(taskId),
20    })
21  }
22 
23  const createMutation = useMutation({
24    mutationFn: createTask,
25    onSuccess: (data) => {
26      queryClient.setQueryData(taskKeys.list({ status }), (old) => [
27        ...old,
28        { id: data.id, title: data.todo, status: 'open' }
29      ])
30    }
31  })
32 
33  const updateMutation = useMutation({
34    mutationFn: updateTask,
35    onMutate: async (updatedTask) => {
36      await queryClient.cancelQueries({ queryKey: taskKeys.list({ status }) })
37      const previousTasks = queryClient.getQueryData(taskKeys.list({ status }))
38      queryClient.setQueryData(taskKeys.list({ status }), (old) =>
39        old.map(t => t.id === updatedTask.id ? { ...t, ...updatedTask } : t)
40      )
41      return { previousTasks }
42    },
43    onError: (error, updatedTask, context) => {
44      queryClient.setQueryData(taskKeys.list({ status }), context.previousTasks)
45    },
46    onSettled: () => {
47      queryClient.invalidateQueries({ queryKey: taskKeys.all })
48    }
49  })
50 
51  const deleteMutation = useMutation({
52    mutationFn: deleteTask,
53    onMutate: async (id) => {
54      await queryClient.cancelQueries({ queryKey: taskKeys.list({ status }) })
55      const previousTasks = queryClient.getQueryData(taskKeys.list({ status }))
56      queryClient.setQueryData(taskKeys.list({ status }), (old) =>
57        old.filter(task => task.id !== id)
58      )
59      return { previousTasks }
60    },
61    onError: (error, id, context) => {
62      queryClient.setQueryData(taskKeys.list({ status }), context.previousTasks)
63    },
64    onSettled: () => {
65      queryClient.invalidateQueries({ queryKey: taskKeys.all })
66    }
67  })
68 
69  if (isLoading) return <p>Loading...</p>
70  if (isError) return <p>Something went wrong.</p>
71 
72  return (
73    <div>
74      <form onSubmit={e => {
75        e.preventDefault()
76        createMutation.mutate({ title })
77        setTitle('')
78      }}>
79        <input value={title} onChange={e => setTitle(e.target.value)} placeholder='New task...' />
80        <button type='submit'>Add</button>
81      </form>
82      <select value={status} onChange={e => setStatus(e.target.value)}>
83        <option value=''>All</option>
84        <option value='open'>Open</option>
85        <option value='inProgress'>In progress</option>
86        <option value='done'>Done</option>
87      </select>
88      <ul>
89        {tasks.map(task => (
90          <li key={task.id} onMouseEnter={() => handleMouseEnter(task.id)}>
91            {task.title}
92            <select value={task.status} onChange={e => updateMutation.mutate({ ...task, status: e.target.value })}>
93              <option value='open'>Open</option>
94              <option value='inProgress'>In progress</option>
95              <option value='done'>Done</option>
96            </select>
97            <button onClick={() => deleteMutation.mutate(task.id)}>Delete</button>
98          </li>
99        ))}
100      </ul>
101    </div>
102  )
103}
```

```javascript
1const BASE = 'https://dummyjson.com/todos'
2 
3export function fetchTasks({ status } = {}) {
4  return fetch(`${BASE}?limit=10`)
5    .then(r => r.json())
6    .then(d => d.todos.map((t, i) => ({
7      id: t.id, title: t.todo,
8      status: t.completed ? 'done' : (i % 3 === 1 ? 'inProgress' : 'open')
9    })))
10    .then(tasks => status ? tasks.filter(t => t.status === status) : tasks)
11}
12 
13export function fetchTask(id) {
14  return fetch(`${BASE}/${id}`).then(res => res.json())
15}
16 
17export function createTask(newTask) {
18  return fetch(`${BASE}/add`, {
19    method: 'POST',
20    headers: { 'Content-Type': 'application/json' },
21    body: JSON.stringify({ todo: newTask.title, completed: false, userId: 1 })
22  }).then(res => res.json())
23}
24 
25export function updateTask(task) {
26  return fetch(`${BASE}/${task.id}`, {
27    method: 'PATCH',
28    headers: { 'Content-Type': 'application/json' },
29    body: JSON.stringify({ completed: task.status === 'done' })
30  }).then(res => res.json())
31}
32 
33export function deleteTask(id) {
34  return fetch(`${BASE}/${id}`, { method: 'DELETE' }).then(res => res.json())
35}
```
