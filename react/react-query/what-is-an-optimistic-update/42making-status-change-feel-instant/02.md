```undefined
1import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
2import { useState } from 'react'
3import { fetchTasks, createTask, updateTask, deleteTask } from './api'
4import { taskKeys } from './queryKeys'
5 
6export default function TaskList() {
7  const [title, setTitle] = useState('')
8  const [status, setStatus] = useState('')
9  const queryClient = useQueryClient()
10 
11  const { data: tasks, isLoading, isError } = useQuery({
12    queryKey: taskKeys.list({ status }),
13    queryFn: () => fetchTasks({ status }),
14  })
15 
16  const createMutation = useMutation({
17    mutationFn: createTask,
18    onSuccess: (data) => {
19      queryClient.setQueryData(taskKeys.list({ status }), (old) => [
20        ...old,
21        { id: data.id, title: data.todo, status: 'open' }
22      ])
23    }
24  })
25 
26  const updateMutation = useMutation({
27    mutationFn: updateTask,
28    onMutate: async (updatedTask) => {
29      await queryClient.cancelQueries({ queryKey: taskKeys.list({ status }) })
30      const previousTasks = queryClient.getQueryData(taskKeys.list({ status }))
31      queryClient.setQueryData(taskKeys.list({ status }), (old) =>
32        old.map(t => t.id === updatedTask.id ? { ...t, ...updatedTask } : t)
33      )
34      return { previousTasks }
35    },
36    onError: (error, updatedTask, context) => {
37      queryClient.setQueryData(taskKeys.list({ status }), context.previousTasks)
38    },
39    onSettled: () => {
40      queryClient.invalidateQueries({ queryKey: taskKeys.all })
41    }
42  })
43 
44  const deleteMutation = useMutation({
45    mutationFn: deleteTask,
46    onSuccess: (_, id) => {
47      queryClient.setQueryData(taskKeys.list({ status }), (old) =>
48        old.filter(t => t.id !== id)
49      )
50    }
51  })
52 
53  if (isLoading) return <p>Loading...</p>
54  if (isError) return <p>Something went wrong.</p>
55 
56  return (
57    <div>
58      <form onSubmit={e => {
59        e.preventDefault()
60        createMutation.mutate({ title })
61        setTitle('')
62      }}>
63        <input value={title} onChange={e => setTitle(e.target.value)} placeholder='New task...' />
64        <button type='submit'>Add</button>
65      </form>
66      <select value={status} onChange={e => setStatus(e.target.value)}>
67        <option value=''>All</option>
68        <option value='open'>Open</option>
69        <option value='inProgress'>In progress</option>
70        <option value='done'>Done</option>
71      </select>
72      <ul>
73        {tasks.map(task => (
74          <li key={task.id}>
75            {task.title}
76            <select value={task.status} onChange={e => updateMutation.mutate({ ...task, status: e.target.value })}>
77              <option value='open'>Open</option>
78              <option value='inProgress'>In progress</option>
79              <option value='done'>Done</option>
80            </select>
81            <button onClick={() => deleteMutation.mutate(task.id)}>Delete</button>
82          </li>
83        ))}
84      </ul>
85    </div>
86  )
87}
```

```javascript
1const BASE = 'https://dummyjson.com/todos'
2 
3export function fetchTasks({ status } = {}) {
4  return fetch(`${BASE}?limit=10`)
5    .then(r => r.json())
6    .then(d => d.todos.map((t, i) => ({
7      id: t.id, title: t.todo,
8      status: t.completed ? 'done' : (i % 3 === 1 ? 'inProgress' : 'open')
9    })))
10    .then(tasks => status ? tasks.filter(t => t.status === status) : tasks)
11}
12 
13export function createTask(newTask) {
14  return fetch(`${BASE}/add`, {
15    method: 'POST',
16    headers: { 'Content-Type': 'application/json' },
17    body: JSON.stringify({ todo: newTask.title, completed: false, userId: 1 })
18  }).then(res => res.json())
19}
20 
21export function updateTask(task) {
22  return fetch(`${BASE}/${task.id}`, {
23    method: 'PATCH',
24    headers: { 'Content-Type': 'application/json' },
25    body: JSON.stringify({ completed: task.status === 'done' })
26  }).then(res => res.json())
27}
28 
29export function deleteTask(id) {
30  return fetch(`${BASE}/${id}`, { method: 'DELETE' }).then(res => res.json())
31}
```
