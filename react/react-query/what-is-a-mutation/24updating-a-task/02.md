```undefined
1import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
2import { useState } from 'react'
3import { fetchTasks, createTask, updateTask, deleteTask } from './api'
4 
5export default function TaskList() {
6  const [title, setTitle] = useState('')
7  const [status, setStatus] = useState('')
8  const queryClient = useQueryClient()
9 
10  const { data: tasks, isLoading, isError } = useQuery({
11    queryKey: ['tasks', { status }],
12    queryFn: () => fetchTasks({ status }),
13  })
14 
15  const createMutation = useMutation({
16    mutationFn: createTask,
17    onSuccess: (data) => {
18      queryClient.setQueryData(['tasks', { status }], (old) => [
19        ...old,
20        { id: data.id, title: data.todo, status: 'open' }
21      ])
22    }
23  })
24 
25  const updateMutation = useMutation({
26    mutationFn: updateTask,
27    onSuccess: (_, updatedTask) => {
28      queryClient.setQueryData(['tasks', { status }], (old) =>
29        old.map(t => t.id === updatedTask.id ? { ...t, ...updatedTask } : t)
30      )
31    }
32  })
33 
34  const deleteMutation = useMutation({
35    mutationFn: deleteTask,
36    onSuccess: (_, id) => {
37      queryClient.setQueryData(['tasks', { status }], (old) =>
38        old.filter(t => t.id !== id)
39      )
40    }
41  })
42 
43  if (isLoading) return <p>Loading...</p>
44  if (isError) return <p>Something went wrong.</p>
45 
46  return (
47    <div>
48      <form onSubmit={e => {
49        e.preventDefault()
50        createMutation.mutate({ title })
51        setTitle('')
52      }}>
53        <input value={title} onChange={e => setTitle(e.target.value)} placeholder='New task...' />
54        <button type='submit'>Add</button>
55      </form>
56      <select value={status} onChange={e => setStatus(e.target.value)}>
57        <option value=''>All</option>
58        <option value='open'>Open</option>
59        <option value='inProgress'>In progress</option>
60        <option value='done'>Done</option>
61      </select>
62      <ul>
63        {tasks.map(task => (
64          <li key={task.id}>
65            {task.title}
66            <select value={task.status} onChange={e => updateMutation.mutate({ ...task, status: e.target.value })}>
67              <option value='open'>Open</option>
68              <option value='inProgress'>In progress</option>
69              <option value='done'>Done</option>
70            </select>
71            <button onClick={() => deleteMutation.mutate(task.id)}>Delete</button>
72          </li>
73        ))}
74      </ul>
75    </div>
76  )
77}
```

```javascript
1const BASE = 'https://dummyjson.com/todos'
2 
3export function fetchTasks({ status } = {}) {
4  return fetch(`${BASE}?limit=10`)
5    .then(r => r.json())
6    .then(d => d.todos.map((t, i) => ({
7      id: t.id, title: t.todo,
8      status: t.completed ? 'done' : (i % 3 === 1 ? 'inProgress' : 'open')
9    })))
10    .then(tasks => status ? tasks.filter(t => t.status === status) : tasks)
11}
12 
13export function createTask(newTask) {
14  return fetch(`${BASE}/add`, {
15    method: 'POST',
16    headers: { 'Content-Type': 'application/json' },
17    body: JSON.stringify({ todo: newTask.title, completed: false, userId: 1 })
18  }).then(res => res.json())
19}
20 
21export function updateTask(task) {
22  return fetch(`${BASE}/${task.id}`, {
23    method: 'PATCH',
24    headers: { 'Content-Type': 'application/json' },
25    body: JSON.stringify({ completed: task.status === 'done' })
26  }).then(res => res.json())
27}
28 
29export function deleteTask(id) {
30  return fetch(`${BASE}/${id}`, { method: 'DELETE' }).then(res => res.json())
31}
```
