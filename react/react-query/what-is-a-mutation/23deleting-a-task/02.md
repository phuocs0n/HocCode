1import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
2import { useState } from 'react'
3import { fetchTasks, createTask, deleteTask } from './api'
4 
5export default function TaskList() {
6  const [title, setTitle] = useState('')
7  const [status, setStatus] = useState('')
8  const queryClient = useQueryClient()
9 
10  const { data: tasks, isLoading, isError } = useQuery({
11    queryKey: ['tasks', { status }],
12    queryFn: () => fetchTasks({ status }),
13  })
14 
15  const createMutation = useMutation({
16    mutationFn: createTask,
17    onSuccess: (data) => {
18      queryClient.setQueryData(['tasks', { status }], (old) => [
19        ...old,
20        { id: data.id, title: data.todo, status: 'open' }
21      ])
22    }
23  })
24 
25  const deleteMutation = useMutation({
26    mutationFn: deleteTask,
27    onSuccess: (_, id) => {
28      queryClient.setQueryData(['tasks', { status }], (old) =>
29        old.filter(t => t.id !== id)
30      )
31    }
32  })
33 
34  if (isLoading) return <p>Loading...</p>
35  if (isError) return <p>Something went wrong.</p>
36 
37  return (
38    <div>
39      <form onSubmit={e => {
40        e.preventDefault()
41        createMutation.mutate({ title })
42        setTitle('')
43      }}>
44        <input value={title} onChange={e => setTitle(e.target.value)} placeholder='New task...' />
45        <button type='submit'>Add</button>
46      </form>
47      <select value={status} onChange={e => setStatus(e.target.value)}>
48        <option value=''>All</option>
49        <option value='open'>Open</option>
50        <option value='inProgress'>In progress</option>
51        <option value='done'>Done</option>
52      </select>
53      <ul>
54        {tasks.map(task => (
55          <li key={task.id}>
56            {task.title}
57            <button onClick={() => deleteMutation.mutate(task.id)}>Delete</button>
58          </li>
59        ))}
60      </ul>
61    </div>
62  )
63}

```javascript
1const BASE = 'https://dummyjson.com/todos'
2 
3export function fetchTasks({ status } = {}) {
4  return fetch(`${BASE}?limit=10`)
5    .then(r => r.json())
6    .then(d => d.todos.map((t, i) => ({
7      id: t.id, title: t.todo,
8      status: t.completed ? 'done' : (i % 3 === 1 ? 'inProgress' : 'open')
9    })))
10    .then(tasks => status ? tasks.filter(t => t.status === status) : tasks)
11}
12 
13export function createTask(newTask) {
14  return fetch(`${BASE}/add`, {
15    method: 'POST',
16    headers: { 'Content-Type': 'application/json' },
17    body: JSON.stringify({ todo: newTask.title, completed: false, userId: 1 })
18  }).then(res => res.json())
19}
20 
21export function deleteTask(id) {
22  return fetch(`${BASE}/${id}`, { method: 'DELETE' }).then(res => res.json())
23}
```
