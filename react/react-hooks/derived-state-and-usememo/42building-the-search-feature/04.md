1import { useRef, useEffect } from "react";

3function Drawer({ entry, onClose }) {
4  const drawerRef = useRef(null);
5
6  useEffect(function () {
7    function handleMouseDown(event) {
8
9      if (drawerRef.current && !drawerRef.current.contains(event.target)) {
10        onClose();
11      }
12    }
13
14    document.addEventListener("mousedown", handleMouseDown);
15    return function () {
16      document.removeEventListener("mousedown", handleMouseDown);
17    };
18  }, [onClose]);
19
20  if (!entry) return null;
21
22  return (
23    <div className="drawer-overlay">
24      <div className="drawer" ref={drawerRef}>
25        <button className="drawer-close" onClick={onClose}>×</button>
26        <h2>{entry.title}</h2>
27        <p className="drawer-meta">{entry.date} · {entry.author}</p>
28        <p className="drawer-content">{entry.content}</p>
29      </div>
30    </div>
31  );
32}
33
34export default Drawer;
