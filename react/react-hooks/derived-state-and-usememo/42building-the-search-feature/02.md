1import { useRef, useEffect } from "react";
2
3function Drawer({ entry, onClose }) {
4  const drawerRef = useRef(null);
5  useEffect(function () {
6    function handleMouseDown(event) {
7      if (drawerRef.current && !drawerRef.current.contains(event.target)) {
8        onClose();
9      }
10    }
11    document.addEventListener("mousedown", handleMouseDown);
12    return function () {
13      document.removeEventListener("mousedown", handleMouseDown);
14    };
15  }, [onClose]);
16  if (!entry) return null;
17
18  return (
19    <div className="drawer-overlay">
20      <div className="drawer" ref={drawerRef}>
21        <button className="drawer-close" onClick={onClose}>×</button>
22        <h2>{entry.title}</h2>
23        <p className="drawer-meta">{entry.date} · {entry.author}</p>
24        <p className="drawer-content">{entry.content}</p>
25      </div>
26    </div>
27  );
28}
