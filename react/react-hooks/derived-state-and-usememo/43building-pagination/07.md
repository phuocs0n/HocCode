3function Drawer({ entry, onClose }) {
4  const drawerRef = useRef(null);
5
6  useEffect(function () {
7    function handleMouseDown(event) {

10      }
11    }
12
13    document.addEventListener("mousedown", handleMouseDown);
14    return function () {
15      document.removeEventListener("mousedown", handleMouseDown);
16    };
17  }, [onClose]);
