## Extract useSearch()

Like before, we'll create a new JS file: /src/hooks/useSearch.js. The hook:

- Accepts the entries array
- Manages query state
- And derives the filtered list

1import { useState, useMemo, useCallback } from "react";
2import Header from "./components/Header";
3import EntryCard from "./components/EntryCard";
4import Drawer from "./components/Drawer";
5import useFetch from "./hooks/useFetch";
6
7const ENTRIES_PER_PAGE = 5;
8
9function App() {
10
11  const { data, isLoading, error } = useFetch(
12    "https://api.programiz.pro/api/Demo/v1/journal-entries"
13  );
14
15  const [selectedEntry, setSelectedEntry] = useState(null);
16
17  const [query, setQuery] = useState("");
18  const [currentPage, setCurrentPage] = useState(1);
19
20  const entries = useMemo(function () {
21    if (!data) return [];
22
23    return data.map(function (post) {
24      return {
25        ...post,
26        date: new Date(post.date).toLocaleDateString("en-US", {
27          year: "numeric", month: "short", day: "numeric"
28        }).toUpperCase()
29      };
30    });
31  }, [data]);
32
33  const filteredEntries = useMemo(function () {
34    if (!query.trim()) return entries;
35    const lower = query.toLowerCase();
36
37    return entries.filter(function (entry) {
38      return (
39        entry.title.toLowerCase().includes(lower) ||
40        entry.content.toLowerCase().includes(lower)
41      );
42    });
43  }, [entries, query]);
44
45  const totalPages = useMemo(function () {
46    return Math.ceil(filteredEntries.length / ENTRIES_PER_PAGE);
47  }, [filteredEntries]);
48
49  const pagedEntries = useMemo(function () {
50    const start = (currentPage - 1) * ENTRIES_PER_PAGE;
51    return filteredEntries.slice(start, start + ENTRIES_PER_PAGE);
52  }, [filteredEntries, currentPage]);
53
54  const handleSelectEntry = useCallback(function (entry) {
55    setSelectedEntry(entry);
56  }, []);
57
58  const handleCloseDrawer = useCallback(function () {
59    setSelectedEntry(null);
60  }, []);
61
62  function handleQueryChange(newQuery) {
63    setQuery(newQuery);
64    setCurrentPage(1);
65  }
66
67  function handleNext() {
68    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
69  }
70
71  function handlePrev() {
72    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
73  }
74
75  return (
76    <div className="main-container">
77      <Header query={query} onQueryChange={handleQueryChange} />
78      <section className="entries-section">
79        <div className="entry-list-header">
80          <h3>Recent Entries</h3>
81          <span className="entry-count">{filteredEntries.length}</span>
82          {query && <span className="entry-count">{"of " + entries.length}</span>}
83        </div>
84        {isLoading ? (
85          <p className="loading-state">Loading entries...</p>
86        ) : error ? (
87          <div className="error-banner">
88            <p className="error-label">Something went wrong.</p>
89            <p className="error-detail">{error}</p>
90          </div>
91        ) : pagedEntries.length === 0 ? (
92          <p className="empty-state">No entries found.</p>
93        ) : (
94          pagedEntries.map((entry) => (
95            <EntryCard
96              key={entry.id}
97              id={entry.id}
98              title={entry.title}
99              date={entry.date}
100              content={entry.content}
101              author={entry.author}
102              onSelect={handleSelectEntry}
103            />
104          ))
105        )}
106        {!isLoading && !error && totalPages > 1 && (
107          <div className="pagination">
108            <button className="btn btn-secondary" onClick={handlePrev} disabled={currentPage === 1}>Previous</button>
109            <span className="pagination-info">Page {currentPage} of {totalPages}</span>
110            <button className="btn btn-secondary" onClick={handleNext} disabled={currentPage === totalPages}>Next</button>
111          </div>
112        )}
113      </section>
114      <Drawer entry={selectedEntry} onClose={handleCloseDrawer} />
115    </div>
116  );
117}
118
119export default App;
```

1import { useState, useEffect } from "react";
2
3function useFetch(url) {
4  const [data, setData] = useState(null);
5  const [isLoading, setIsLoading] = useState(true);
6  const [error, setError] = useState(null);
7
8  useEffect(function () {
9    async function fetchData() {
10      setIsLoading(true);
11      setError(null);
12
13      try {
14        const res = await fetch(url);
15        if (!res.ok) {
16          throw new Error("Could not load data. Status: " + res.status);
17        }
18        const json = await res.json();
19        setData(json);
20        setIsLoading(false);
21      }
22
23      catch (err) {
24        setError(err.message);
25        setIsLoading(false);
26      }
27    }
28
29    fetchData();
30  }, [url]);
31
32  return { data, isLoading, error };
33}
34
35export default useFetch;
```

3function useSearch(entries) {
4  const [query, setQuery] = useState("");
5
6  const filteredEntries = useMemo(function () {
7    if (!query.trim()) return entries;
8    const lower = query.toLowerCase();
9

12        entry.title.toLowerCase().includes(lower) ||

17
18  return { query, setQuery, filteredEntries };
19}

How It Works

- useSearch() accepts entries as a parameter.
- useSearch() doesn't fetch; it only filters.
- The caller can use the returned query, setQuery(), and filteredEntries however it needs.
