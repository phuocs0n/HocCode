## The Problem: Search Logic Still in App

After extracting useFetch(), the search feature is still in App directly.

There's a useState() for query and a useMemo() for filteredEntries. They always travel together — any component that needed search would need both:

1import { useState, useEffect } from "react";

3function useFetch(url) {
4  const [data, setData] = useState(null);
5  const [isLoading, setIsLoading] = useState(true);
6  const [error, setError] = useState(null);
7
8  useEffect(function () {
9    async function fetchData() {
10      setIsLoading(true);
11      setError(null);
12
13      try {
14        const res = await fetch(url);
15        if (!res.ok) {
16          throw new Error("Could not load data. Status: " + res.status);
17        }
18        const json = await res.json();
19        setData(json);
20        setIsLoading(false);
21      }
22
23      catch (err) {
24        setError(err.message);
25        setIsLoading(false);
26      }
27    }
28
29    fetchData();
30  }, [url]);
31
32  return { data, isLoading, error };
33}
