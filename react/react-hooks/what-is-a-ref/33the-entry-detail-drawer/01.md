## Building the Drawer Component

We'll start by creating a new file: /src/components/Drawer.jsx.

Drawer receives two props:

1. entry (the selected entry object, or null when nothing is selected)
2. onClose (called when the drawer should close)

```undefined
1import { useState, useEffect } from "react";
2import Header from "./components/Header";
3import EntryCard from "./components/EntryCard";
4import Drawer from "./components/Drawer";
5
6function App() {
7  const [entries, setEntries] = useState([]);
8  const [isLoading, setIsLoading] = useState(true);
9  const [error, setError] = useState(null);
10  const [refreshKey, setRefreshKey] = useState(0);
11  const [selectedEntry, setSelectedEntry] = useState(null);
12
13  useEffect(function () {
14    async function loadEntries() {
15      setIsLoading(true);
16      setError(null);
17      try {
18        const res = await fetch("https://api.programiz.pro/api/Demo/v1/journal-entries");
19        
20        if (!res.ok) {
21          throw new Error("Could not load entries. Status: " + res.status);
22        }
23        
24        const data = await res.json();
25        const mapped = data.map((post) => {
26          return {
27            ...post,
28            date: new Date(post.date).toLocaleDateString("en-US", {
29              year: "numeric", month: "short", day: "numeric"
30            }).toUpperCase()
31          };
32        });
33        setEntries(mapped);
34        setIsLoading(false);
35      }
36      catch (err) {
37        setError(err.message);
38        setIsLoading(false);
39      }
40    }
41    loadEntries();
42  }, [refreshKey]);
43
44  function handleSelectEntry(entry) {
45    setSelectedEntry(entry);
46  }
47
48  return (
49    <div className="main-container">
50      <Header />
51      <section className="entries-section">
52        <div className="entry-list-header">
53          <h3>Recent Entries</h3>
54          <span className="entry-count">{entries.length}</span>
55          <button
56            className="btn btn-secondary"
57            onClick={() => setRefreshKey((k) => k + 1)}
58          >
59            Refresh
60          </button>
61        </div>
62        {isLoading ? (
63          <p className="loading-state">Loading entries...</p>
64        ) : error ? (
65          <div className="error-banner">
66            <p className="error-label">Something went wrong.</p>
67            <p className="error-detail">{error}</p>
68          </div>
69        ) : entries.length === 0 ? (
70          <p className="empty-state">No entries found.</p>
71        ) : (
72          entries.map((entry) => (
73            <EntryCard
74              key={entry.id}
75              id={entry.id}
76              title={entry.title}
77              date={entry.date}
78              content={entry.content}
79              author={entry.author}
80              onSelect={handleSelectEntry}
81            />
82          ))
83        )}
84      </section>
85      <Drawer
86        entry={selectedEntry}
87        onClose={() => setSelectedEntry(null)}
88      />
89    </div>
90  );
91}
92
93export default App;
```

How the Code Works

1. When entry is null, the component returns null and nothing renders.
2. When an entry is selected, the drawer appears over the page with an overlay behind it.

If you run the code, you should see that Drawer.jsx has now been created. But it won't do anything since we haven't updated our App and EntryCard.
