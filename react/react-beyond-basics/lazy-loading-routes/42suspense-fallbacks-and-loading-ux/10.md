## Use SkeletonPage as the Route Fallback

In App.jsx, we'll import SkeletonPage as a regular static import and use it as the <Suspense> fallback for every route.

Skeleton loaders are always small and fast, so they should never be lazy-loaded themselves.

```react
import SkeletonPage from './components/SkeletonPage'
```

Then, we'll ensure each route has its own <Suspense> boundary inline in the element prop, with SkeletonPage as the fallback.

```undefined
1<Route path="/" element={<Suspense fallback={<SkeletonPage />}><HomePage /></Suspense>} />
2
3<Route path="/entries/:id" element={<Suspense fallback={<SkeletonPage />}><EntryDetailPage /></Suspense>} />
4
5<Route path="*" element={<Suspense fallback={<SkeletonPage />}><NotFoundPage /></Suspense>} />
```

```javascript
24        setIsLoading(false);

28  }, [url, refetchKey]);
29
30  function refetch() {
31    setRefetchKey(function (k) { return k + 1; });
```

```javascript
1import { useState, useMemo } from "react";
2
3function usePagination(entries, pageSize) {
4  const [currentPage, setCurrentPage] = useState(1);
5
6  const totalPages = useMemo(function () {
7    return Math.ceil(entries.length / pageSize);
8  }, [entries, pageSize]);
9
10  const pagedEntries = useMemo(function () {
11    const start = (currentPage - 1) * pageSize;
12    return entries.slice(start, start + pageSize);
13  }, [entries, currentPage, pageSize]);
14
15  function handleNext() {
16    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
17  }
18
19  function handlePrev() {
20    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
21  }
22
23  function reset() {
24    setCurrentPage(1);
25  }
26
27  return { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset };
28}
29
30export default usePagination
```

```undefined
1import { createContext, useReducer, useState, useContext, useEffect } from 'react'
2
3const JournalContext = createContext(null)
4const API_URL = 'https://dev.api.programiz.pro/api/Demo/v1/journal-entries'
5
6function journalReducer(state, action) {
7  switch (action.type) {
8    case 'SET_ENTRIES':
9      return action.payload
10    case 'BOOKMARK_ENTRY':
11      return state.map(function (entry) {
12        return entry.id === action.payload.id
13          ? { ...entry, bookmarked: !entry.bookmarked }
14          : entry
15      })
16    default:
17      return state
18  }
19}
20
21function JournalProvider({ children }) {
22  const [entries, dispatch] = useReducer(journalReducer, [])
23  const [isLoading, setIsLoading] = useState(true)
24  const [error, setError] = useState(null)
25  const [refetchKey, setRefetchKey] = useState(0)
26
27  useEffect(function () {
28    async function loadEntries() {
29      setIsLoading(true)
30      setError(null)
31      try {
32        const res = await fetch(API_URL)
33        if (!res.ok) throw new Error('Could not load data. Status: ' + res.status)
34        const data = await res.json()
35        const formatted = data.map(function (entry) {
36          return {
37            ...entry,
38            bookmarked: false,
39            date: new Date(entry.date).toLocaleDateString('en-US', {
40              year: 'numeric', month: 'short', day: 'numeric'
41            }).toUpperCase()
42          }
43        })
44        dispatch({ type: 'SET_ENTRIES', payload: formatted })
45      } catch (err) {
46        setError(err.message)
47      } finally {
48        setIsLoading(false)
49      }
50    }
51    loadEntries()
52  }, [refetchKey])
53
54  function refetch() {
55    setRefetchKey(function (k) { return k + 1 })
56  }
57
58  return (
59    <JournalContext.Provider value={{ entries, dispatch, isLoading, error, refetch }}>
60      {children}
61    </JournalContext.Provider>
62  )
63}
64
65function useJournal() {
66  return useContext(JournalContext)
67}
68
69export { JournalProvider, useJournal }
```
