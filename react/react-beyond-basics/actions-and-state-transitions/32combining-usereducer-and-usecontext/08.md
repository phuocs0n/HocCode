21  }
22

24    setCurrentPage(1);
25  }
26

1import { createContext, useReducer, useState, useContext, useEffect } from 'react'
2
3const JournalContext = createContext(null)
4
5const API_URL = 'https://dev.api.programiz.pro/api/Demo/v1/journal-entries'
6
7function journalReducer(state, action) {
8  switch (action.type) {
9    case 'SET_ENTRIES':
10      return action.payload
11    case 'BOOKMARK_ENTRY':
12      return state.map(function (entry) {
13        return entry.id === action.payload.id
14          ? { ...entry, bookmarked: !entry.bookmarked }
15          : entry
16      })
17    default:
18      return state
19  }
20}
21
22function JournalProvider({ children }) {
23  const [entries, dispatch] = useReducer(journalReducer, [])
24  const [isLoading, setIsLoading] = useState(true)
25  const [error, setError] = useState(null)
26  const [refetchKey, setRefetchKey] = useState(0)
27
28  useEffect(function () {
29    async function loadEntries() {
30      setIsLoading(true)
31      setError(null)
32      try {
33        const res = await fetch(API_URL)
34        if (!res.ok) throw new Error('Could not load data. Status: ' + res.status)
35        const data = await res.json()
36        const formatted = data.map(function (entry) {
37          return {
38            ...entry,
39            bookmarked: false,
40            date: new Date(entry.date).toLocaleDateString('en-US', {
41              year: 'numeric', month: 'short', day: 'numeric'
42            }).toUpperCase()
43          }
44        })
45        dispatch({ type: 'SET_ENTRIES', payload: formatted })
46      } catch (err) {
47        setError(err.message)
48      } finally {
49        setIsLoading(false)
50      }
51    }
52    loadEntries()
53  }, [refetchKey])
54
55  function refetch() {
56    setRefetchKey(function (k) { return k + 1 })
57  }
58
59  return (
60    <JournalContext.Provider value={{ entries, dispatch, isLoading, error, refetch }}>
61      {children}
62    </JournalContext.Provider>
63  )
64}
65
66function useJournal() {
67  return useContext(JournalContext)
68}
69
70export { JournalProvider, useJournal }
