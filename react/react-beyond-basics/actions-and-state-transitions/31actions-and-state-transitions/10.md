## Updating HomePage with useReducer()

We'll replace the data returned by useFetch() with an explicit useReducer().

The fetch effect now dispatches SET_ENTRIES when data arrives. The behavior is identical to before:

- The fetch happens on mount
- Entries appear like before
- Search and pagination work the same

But now the state transition is named.

Anyone reading this code can see that SET_ENTRIES is an event with a specific meaning, not just an anonymous setEntries(data) call.

They can also see that another action (BOOKMARK_ENTRY) is present in the code. We'll add the button for it on the next page.

For now, click Run Code and you'll see that functionally, nothing has changed.

3function useFetch(url) {
4  const [data, setData] = useState(null);
5  const [isLoading, setIsLoading] = useState(true);
6  const [error, setError] = useState(null);
7  const [refetchKey, setRefetchKey] = useState(0);
8
9  useEffect(function () {
10    async function fetchData() {
11      setIsLoading(true);
12      setError(null);
13      try {
14        const res = await fetch(url);
15        
16        if (!res.ok) {
17          throw new Error("Could not load data. Status: " + res.status);
18        }
19        const json = await res.json();
20        setData(json);
21        setIsLoading(false);
22      } catch (err) {
23        setError(err.message);
24        setIsLoading(false);
25      }
26    }
27    fetchData();
28  }, [url, refetchKey]);
29
30  function refetch() {
31    setRefetchKey(function (k) { return k + 1; });
32  }
33
34  return { data, isLoading, error, refetch };
35}
36
37export default useFetch;

3function usePagination(entries, pageSize) {
4  const [currentPage, setCurrentPage] = useState(1);
5
6  const totalPages = useMemo(function () {
7    return Math.ceil(entries.length / pageSize);
8  }, [entries, pageSize]);
9
10  const pagedEntries = useMemo(function () {
11    const start = (currentPage - 1) * pageSize;
12    return entries.slice(start, start + pageSize);
13  }, [entries, currentPage, pageSize]);
14
15  function handleNext() {
16    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
17  }
18
19  function handlePrev() {
20    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
21  }
22
23  function reset() {
24    setCurrentPage(1);
25  }
26
27  return { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset };
28}

3const ThemeContext = createContext(null)
4
5export function ThemeProvider({ children }) {
6  const [theme, setTheme] = useState(localStorage.getItem('theme') || 'light')
7
8  useEffect(function () {
9    document.documentElement.setAttribute('data-theme', theme)
10    localStorage.setItem('theme', theme)
11  }, [theme])
12
13  function toggleTheme() {
14    setTheme(function (t) { return t === 'light' ? 'dark' : 'light' })
15  }
16
17  return (
18    <ThemeContext.Provider value={{ theme, toggleTheme }}>
19      {children}
20    </ThemeContext.Provider>
21  )
22}
