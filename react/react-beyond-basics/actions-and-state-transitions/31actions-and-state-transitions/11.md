## Add the Bookmark Button

The code to mark an entry as bookmarked is already present in our components. So all that remains is adding the bookmark button to each entry in EntryCard.jsx.

This bookmark button will behave in the following way:

- When an entry is not bookmarked, the button text will say ☆ Bookmark.
- When an entry is bookmarked, the button text will say ★ Bookmarked.

Additionally, we'll wrap this button inside a new <div> dedicated to actions related to the entry card.

This makes it easier for you to personally add other actions of your choosing in the future (such as adding a Mark as Read button).

Now, let's code all this in EntryCard.jsx:

```javascript
1import { useState, useMemo } from "react";
2
3function useSearch(entries) {
4  const [query, setQuery] = useState("");
5
6  const filteredEntries = useMemo(function () {
7    if (!query.trim()) return entries;
8    const lower = query.toLowerCase();
9    return entries.filter(function (entry) {
10      return (
11        entry.title.toLowerCase().includes(lower) ||
12        entry.content.toLowerCase().includes(lower)
13      );
14    });
15  }, [entries, query]);
16
17  return { query, setQuery, filteredEntries };
18}
19
20export default useSearch;
```

```javascript
1import { useState, useMemo } from "react";
2
3function usePagination(entries, pageSize) {
4  const [currentPage, setCurrentPage] = useState(1);
5
6  const totalPages = useMemo(function () {
7    return Math.ceil(entries.length / pageSize);
8  }, [entries, pageSize]);
9
10  const pagedEntries = useMemo(function () {
11    const start = (currentPage - 1) * pageSize;
12    return entries.slice(start, start + pageSize);
13  }, [entries, currentPage, pageSize]);
14
15  function handleNext() {
16    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
17  }
18
19  function handlePrev() {
20    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
21  }
22
23  function reset() {
24    setCurrentPage(1);
25  }
26
27  return { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset };
28}
29
30export default usePagination;
```

```undefined
1import { useReducer, useState, useEffect } from 'react'
2import Header from "../components/Header";
3import EntryCard from "../components/EntryCard";
4import useSearch from "../hooks/useSearch";
5import usePagination from "../hooks/usePagination";

7const ENTRIES_PER_PAGE = 5;
8const API_URL = 'https://dev.api.programiz.pro/api/Demo/v1/journal-entries'
9
10function journalReducer(state, action) {
11  switch (action.type) {
12    case 'SET_ENTRIES':
13      return action.payload.map(function (entry) {
14        return { ...entry, bookmarked: false }
15      })
16    case 'BOOKMARK_ENTRY':
17      return state.map(function (entry) {
18        return entry.id === action.payload.id
19          ? { ...entry, bookmarked: !entry.bookmarked }
20          : entry
21      })
22    default:
23      return state
24  }
25}
26
27function HomePage() {
28  const [entries, dispatch] = useReducer(journalReducer, [])
29  const [isLoading, setIsLoading] = useState(true)
30  const [error, setError] = useState(null)
31  const [refetchKey, setRefetchKey] = useState(0)
32
33  const { query, setQuery, filteredEntries } = useSearch(entries)
34  const { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset } = usePagination(
35    filteredEntries,
36    ENTRIES_PER_PAGE
37  )
38
39  useEffect(function () {
40    async function loadEntries() {
41      setIsLoading(true)
42      setError(null)
43      try {
44        const res = await fetch(API_URL)
45        if (!res.ok) throw new Error('Could not load data. Status: ' + res.status)
46        const data = await res.json()
47        dispatch({ type: 'SET_ENTRIES', payload: data })
48      } catch (err) {
49        setError(err.message)
50      } finally {
51        setIsLoading(false)
52      }
53    }
54    loadEntries()
55  }, [refetchKey])
56
57  function refetch() {
58    setRefetchKey(function (k) { return k + 1 })
59  }
60
61  function handleQueryChange(newQuery) {
62    setQuery(newQuery)
63    reset()
64  }
65
66  return (

68      <Header query={query} onQueryChange={handleQueryChange} />
69      <section className="entries-section">

71          <h3>Recent Entries</h3>
72          <span className="entry-count">{filteredEntries.length}</span>
73          {query && <span className="entry-count">{"of " + entries.length}</span>}
74          <button className="btn btn-secondary" onClick={refetch}>Refresh</button>
75        </div>
76        {isLoading ? (
77          <p className="loading-state">Loading entries...</p>
78        ) : error ? (
79          <div className="error-banner">
80            <p className="error-label">Something went wrong.</p>
81            <p className="error-detail">{error}</p>
82          </div>
83        ) : pagedEntries.length === 0 ? (
84          <p className="empty-state">No entries found.</p>
85        ) : (
86          pagedEntries.map((entry) => (
87            <EntryCard
88              key={entry.id}
89              id={entry.id}
90              title={entry.title}
91              date={entry.date}
92              content={entry.content}
93              author={entry.author}
94              bookmarked={entry.bookmarked}
95              onBookmark={function () { dispatch({ type: 'BOOKMARK_ENTRY', payload: { id: entry.id } }) }}
96            />
97          ))
98        )}
99        {!isLoading && !error && totalPages > 1 && (
100          <div className="pagination">
101            <button className="btn btn-secondary" onClick={handlePrev} disabled={currentPage === 1}>Previous</button>
102            <span className="pagination-info">Page {currentPage} of {totalPages}</span>
103            <button className="btn btn-secondary" onClick={handleNext} disabled={currentPage === totalPages}>Next</button>
104          </div>
105        )}
```

```undefined
5export function ThemeProvider({ children }) {
6  const [theme, setTheme] = useState(localStorage.getItem('theme') || 'light')
7

10    localStorage.setItem('theme', theme)

13  function toggleTheme() {
14    setTheme(function (t) { return t === 'light' ? 'dark' : 'light' })
15  }
16
17  return (
18    <ThemeContext.Provider value={{ theme, toggleTheme }}>
19      {children}
20    </ThemeContext.Provider>
21  )
```
