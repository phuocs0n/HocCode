## Passing Theme as Props

Here's the approach: App.jsx owns the theme state and passes it down to every component that needs it.

```react
import { useState } from "react";
import { Routes, Route } from "react-router-dom";
import HomePage from "./pages/HomePage";
import EntryDetailPage from "./pages/EntryDetailPage";
import NotFoundPage from "./pages/NotFoundPage";

function App() {
  const [theme, setTheme] = useState("light");
    
  function toggleTheme() {
    setTheme(function (t) { return t === "light" ? "dark" : "light";          
});
  }

  return (
    <div className="main-container">
      <Routes>
        <Route path="/" element={<HomePage theme={theme} toggleTheme={toggleTheme} />} />
        <Route path="/entries/:id" element={<EntryDetailPage theme={theme} />} />
        <Route path="*" element={<NotFoundPage />} />
      </Routes>
    </div>
  );
}

export default App;
```

This works, but it already feels heavy.

And we haven't even started passing theme deeper, from HomePage down to Header, or from EntryDetailPage down to whatever components live inside it.

This pattern has a name. Let's look at why it becomes a problem.
