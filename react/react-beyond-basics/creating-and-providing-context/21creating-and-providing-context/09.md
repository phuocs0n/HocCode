## Wrapping the App with ThemeProvider

We'll add ThemeProvider to index.jsx, the app's entry point, right alongside BrowserRouter. Every component in the tree will now have access to the theme.

Here's how index.jsx will look:

```javascript
3function useFetch(url) {
4  const [data, setData] = useState(null);
5  const [isLoading, setIsLoading] = useState(true);
6  const [error, setError] = useState(null);
7  const [refetchKey, setRefetchKey] = useState(0);

10    async function fetchData() {
11      setIsLoading(true);
12      setError(null);
13      try {
14        const res = await fetch(url);
15        
16        if (!res.ok) {
17          throw new Error("Could not load data. Status: " + res.status);
18        }
19        const json = await res.json();
20        setData(json);
21        setIsLoading(false);
22      } catch (err) {
23        setError(err.message);
24        setIsLoading(false);
25      }
26    }
27    fetchData();
28  }, [url, refetchKey]);
29
30  function refetch() {
31    setRefetchKey(function (k) { return k + 1; });
32  }
33
34  return { data, isLoading, error, refetch };
35}
```

```javascript
1import { useState, useMemo } from "react";
2
3function useSearch(entries) {
4  const [query, setQuery] = useState("");
5
6  const filteredEntries = useMemo(function () {
7    if (!query.trim()) return entries;
8    const lower = query.toLowerCase();
9
10    return entries.filter(function (entry) {
11      return (
12        entry.title.toLowerCase().includes(lower) ||
13        entry.content.toLowerCase().includes(lower)
14      );
15    });
16  }, [entries, query]);
17
18  return { query, setQuery, filteredEntries };
19}
20
21export default useSearch;
```

```javascript
1import { useState, useMemo } from "react";
2
3function usePagination(entries, pageSize) {
4  const [currentPage, setCurrentPage] = useState(1);
5
6  const totalPages = useMemo(function () {
7    return Math.ceil(entries.length / pageSize);
8  }, [entries, pageSize]);
9
10  const pagedEntries = useMemo(function () {
11    const start = (currentPage - 1) * pageSize;
12    return entries.slice(start, start + pageSize);
13  }, [entries, currentPage, pageSize]);
14
15  function handleNext() {
16    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
17  }
18
19  function handlePrev() {
20    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
21  }
22
23  function reset() {
24    setCurrentPage(1);
25  }
26
27  return { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset };
28}
29
30export default usePagination;
```

```undefined
1import { createContext, useContext, useState } from "react";
2
3const ThemeContext = createContext(null);
4
5export function ThemeProvider({ children }) {
6  const [theme, setTheme] = useState("light");
7  function toggleTheme() {
8    setTheme(function (t) { return t === "light" ? "dark" : "light"; });
9  }
10
11  return (
12    <ThemeContext.Provider value={{ theme, toggleTheme }}>
13      {children}
14    </ThemeContext.Provider>
15  );
16}
17
18export function useTheme() {
19  return useContext(ThemeContext);
20}
```

With ThemeProvider in place, the theme is available to any component in the tree.

Check App.jsx and notice that there's no change at all.

There's no theme prop, no toggleTheme() prop, no useState() for theme. App.jsx stays focused on layout and routing.

When Header needs theme and toggleTheme(), it will call useTheme() directly. We'll do that in the next lesson.
