```javascript
26    }

28  }, [url, refetchKey]);
```

```javascript
1import { useState, useMemo } from "react";
2
3function usePagination(entries, pageSize) {
4  const [currentPage, setCurrentPage] = useState(1);
5
6  const totalPages = useMemo(function () {
7    return Math.ceil(entries.length / pageSize);
8  }, [entries, pageSize]);
9
10  const pagedEntries = useMemo(function () {
11    const start = (currentPage - 1) * pageSize;
12    return entries.slice(start, start + pageSize);
13  }, [entries, currentPage, pageSize]);
14
15  function handleNext() {
16    setCurrentPage(function (p) { return Math.min(p + 1, totalPages); });
17  }
18
19  function handlePrev() {
20    setCurrentPage(function (p) { return Math.max(p - 1, 1); });
21  }
22
23  function reset() {
24    setCurrentPage(1);
25  }
26
27  return { currentPage, pagedEntries, totalPages, handleNext, handlePrev, reset };
28}
29
30export default usePagination;
```

Here's what happens when the user clicks the button:

1. toggleTheme() is called
2. setTheme() runs inside the Provider
3. The Provider's state updates
4. The new theme value propagates to every consumer in the tree

The button label flips automatically because it reads from the same theme value.

Notice that no prop was passed from App or through any intermediate component. Instead, Header reached straight into ThemeContext and got exactly what it needed.

But visually, nothing changes yet. We need to make the DOM respond to the theme value.
