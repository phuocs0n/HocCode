4export default function EntryList() {

8  const searchQuery = useJournalStore((state) => state.searchQuery)
9 
10  const filteredEntries = searchQuery.trim()
11    ? entries.filter(
12        (entry) =>
13          entry.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
14          entry.content.toLowerCase().includes(searchQuery.toLowerCase()) ||
15          entry.date.toLowerCase().includes(searchQuery.toLowerCase())
16      )
17    : entries
18 
19  if (isLoading) return <p className="loading">Loading entries...</p>
20  if (error) return <p className="error">Something went wrong: {error}</p>
21 
22  return (
23    <div className="entry-list">
24      <p className="entry-count">
25        {filteredEntries.length}{' '}
26        {filteredEntries.length === 1 ? 'entry' : 'entries'}
27      </p>
28      {filteredEntries.length === 0 ? (
29        <p className="empty-state">No entries found.</p>
30      ) : (
31        filteredEntries.map((entry) => (
32          <EntryCard key={entry.id} entry={entry} />
33        ))
34      )}
35    </div>
36  )
37}

1import { useState } from 'react'
2import useJournalStore from '../store/useJournalStore'
3 
4export default function EntryCard({ entry }) {
5  const [isEditing, setIsEditing] = useState(false)
6  const [editTitle, setEditTitle] = useState(entry.title)
7  const [editContent, setEditContent] = useState(entry.content)
8 
9  const editEntry = useJournalStore((state) => state.editEntry)
10  const deleteEntry = useJournalStore((state) => state.deleteEntry)
11 
12  function handleSave() {
13    if (!editTitle.trim() || !editContent.trim()) return
14    editEntry(entry.id, editTitle, editContent)
15    setIsEditing(false)
16  }
17 
18  function handleCancel() {
19    setEditTitle(entry.title)
20    setEditContent(entry.content)
21    setIsEditing(false)
22  }
23 
24  if (isEditing) {
25    return (
26      <div className="entry-card">
27        <input className="input-field" type="text" value={editTitle} onChange={(e) => setEditTitle(e.target.value)} />
28        <textarea className="textarea-field" value={editContent} onChange={(e) => setEditContent(e.target.value)} rows={4} />
29        <div className="entry-actions">
30          <button className="btn-save" onClick={handleSave}>Save</button>
31          <button className="btn-cancel" onClick={handleCancel}>Cancel</button>
32        </div>
33      </div>
34    )
35  }
36 
37  return (
38    <div className="entry-card">
39      <p className="entry-date">{entry.date}</p>
40      <h2 className="entry-title">{entry.title}</h2>
41      <p className="entry-content">{entry.content}</p>
42      <div className="entry-actions">
43        <button className="btn-edit" onClick={() => setIsEditing(true)}>Edit</button>
44        <button className="btn-delete" onClick={() => deleteEntry(entry.id)}>Delete</button>
45      </div>

Here is what is happening. Every time theme or searchQuery changes Zustand saves them to localStorage. When the page reloads it reads them back before the app renders. So dark mode stays dark. The search stays set. And the entries are fetched fresh from the API just like always.
